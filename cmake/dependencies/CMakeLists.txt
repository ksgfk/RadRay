cmake_minimum_required(VERSION 3.25)
project(RadRayDependency LANGUAGES C CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0126 NEW)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
set(CMAKE_EXPORT_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
if (RADRAY_DEPENDENCY_TRIPLET MATCHES "^windows-x64-(msvc|clangcl)(-staticcrt)?$")
    if (NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64" OR
        (RADRAY_DEPENDENCY_TRIPLET MATCHES "^windows-x64-msvc" AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC") OR
        (RADRAY_DEPENDENCY_TRIPLET MATCHES "^windows-x64-clangcl" AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang"))
        message(FATAL_ERROR "The selected compiler does not match ${RADRAY_DEPENDENCY_TRIPLET}; use an x64 toolchain")
    endif()
elseif(RADRAY_DEPENDENCY_TRIPLET STREQUAL "macos-arm64-appleclang")
    if (NOT APPLE OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang" OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
        message(FATAL_ERROR "macos-arm64-appleclang requires AppleClang targeting arm64")
    endif()
else()
    message(FATAL_ERROR "Unsupported dependency triplet: ${RADRAY_DEPENDENCY_TRIPLET}")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../RuntimeLibrary.cmake")
radray_check_repository_crt("${RADRAY_DEPENDENCY_TRIPLET}")
if (MSVC)
    add_compile_options(/utf-8)
endif()
if (NOT EXISTS "${RADRAY_DEPENDENCY_SOURCE}")
    message(FATAL_ERROR "Missing restored source: ${RADRAY_DEPENDENCY_SOURCE}")
endif()
if (RADRAY_DEPENDENCY_NAME STREQUAL "libjpeg")
    include(libjpeg.cmake)
elseif(RADRAY_DEPENDENCY_NAME STREQUAL "stdexec")
    find_package(Threads REQUIRED)
    add_library(stdexec INTERFACE)
    set_target_properties(stdexec PROPERTIES EXPORT_NAME stdexec)
    target_compile_features(stdexec INTERFACE cxx_std_20)
    target_link_libraries(stdexec INTERFACE Threads::Threads)
    target_include_directories(stdexec INTERFACE "$<INSTALL_INTERFACE:include>")
    install(DIRECTORY "${RADRAY_DEPENDENCY_SOURCE}/include/" DESTINATION include)
    install(TARGETS stdexec EXPORT stdexec-targets)
    install(EXPORT stdexec-targets NAMESPACE STDEXEC:: DESTINATION lib/cmake/stdexec)
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/stdexecConfig.cmake" CONTENT
        "include(CMakeFindDependencyMacro)\nfind_dependency(Threads)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/stdexec-targets.cmake\")\n" @ONLY)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/stdexecConfig.cmake" DESTINATION lib/cmake/stdexec)
else()
    if (RADRAY_DEPENDENCY_NAME STREQUAL "volk")
        find_package(VulkanHeaders CONFIG REQUIRED)
        get_target_property(includes Vulkan::Headers INTERFACE_INCLUDE_DIRECTORIES)
        include_directories(${includes})
    endif()
    add_subdirectory("${RADRAY_DEPENDENCY_SOURCE}" upstream)
    if (RADRAY_DEPENDENCY_NAME STREQUAL "zlib" AND WIN32)
        # FindZLIB supports this spelling without a package-specific static hint.
        set_target_properties(zlibstatic PROPERTIES OUTPUT_NAME zlibstatic)
    endif()
endif()
