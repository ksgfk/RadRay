cmake_minimum_required(VERSION 3.25)
project(vulkan_present_repro LANGUAGES C)
if(NOT WIN32)
    message(FATAL_ERROR "This reproducer uses Win32 WSI")
endif()
find_package(Vulkan REQUIRED)
add_executable(vulkan_present_repro main.c)
set_target_properties(vulkan_present_repro PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_definitions(vulkan_present_repro PRIVATE VK_USE_PLATFORM_WIN32_KHR WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(vulkan_present_repro PRIVATE Vulkan::Vulkan user32)
if(MSVC)
    target_compile_options(vulkan_present_repro PRIVATE /W4)
endif()
